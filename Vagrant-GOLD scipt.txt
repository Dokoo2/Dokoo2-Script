local ThemeRefs = { bg = {}, panel = {}, stroke = {}, button = {}, text = {}, accent = {} }
local SyncCallbacks = {}

local function registerSync(fn) table.insert(SyncCallbacks, fn); fn() end
local function syncAll() for _, fn in ipairs(SyncCallbacks) do fn() end end

local function addBg(el) table.insert(ThemeRefs.bg, {el=el, p="BackgroundColor3"}); el.BackgroundColor3 = Theme.Background end
local function addPanel(el) table.insert(ThemeRefs.panel, {el=el, p="BackgroundColor3"}); el.BackgroundColor3 = Theme.Panel end
local function addStroke(el) table.insert(ThemeRefs.stroke, {el=el, p="Color"}); el.Color = Theme.Outline end
local function addBtn(el) table.insert(ThemeRefs.button, {el=el, p="BackgroundColor3"}); el.BackgroundColor3 = Theme.Button end
local function addText(el) table.insert(ThemeRefs.text, {el=el, p="TextColor3"}); el.TextColor3 = Theme.Text end
local function addAccent(el) table.insert(ThemeRefs.accent, {el=el, p="BackgroundColor3"}); el.BackgroundColor3 = Theme.Accent end

local function applyTheme()
    for _, it in ipairs(ThemeRefs.bg) do it.el[it.p] = Theme.Background end
    for _, it in ipairs(ThemeRefs.panel) do it.el[it.p] = Theme.Panel end
    for _, it in ipairs(ThemeRefs.stroke) do it.el[it.p] = Theme.Outline end
    for _, it in ipairs(ThemeRefs.button) do it.el[it.p] = Theme.Button end
    for _, it in ipairs(ThemeRefs.text) do it.el[it.p] = Theme.Text end
    for _, it in ipairs(ThemeRefs.accent) do it.el[it.p] = Theme.Accent end
    syncAll()
end

local function applyCursorImage(id)
    if not id or id == "" then
        if cursorImage then cursorImage.Image = "" end
        CursorCfg.ImageId = ""; return
    end
    local cleanId = tostring(id):gsub("%s", "")
    local url
    if cleanId:find("rbxassetid://") then url = cleanId
    elseif cleanId:find("^%d+$") then url = "rbxassetid://" .. cleanId
    else url = cleanId end
    CursorCfg.ImageId = url
    if cursorImage then cursorImage.Image = url end
end

local espHighlights, espBillboards = {}, {}

local function cleanupESPForPlayer(player)
    if espHighlights[player] then espHighlights[player]:Destroy(); espHighlights[player] = nil end
    if espBillboards[player] then espBillboards[player]:Destroy(); espBillboards[player] = nil end
end

local function colorToRichTag(c)
    return string.format("rgb(%d,%d,%d)",
        math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5))
end

local function buildESPLabelText(player, humanoid, distance)
    local lines = {}
    if Config.ESP.ShowName then
        table.insert(lines, string.format('<font color="%s">%s</font>', colorToRichTag(Config.ESP.NameColor or Config.ESP.TextColor), player.Name))
    end
    if Config.ESP.ShowHP then
        table.insert(lines, string.format('<font color="%s">HP: %d/%d</font>',
            colorToRichTag(Config.ESP.HPColor or Config.ESP.TextColor), math.floor(humanoid.Health), math.floor(humanoid.MaxHealth)))
    end
    if Config.ESP.ShowDistance then
        table.insert(lines, string.format('<font color="%s">Distance: %d studs</font>',
            colorToRichTag(Config.ESP.DistanceColor or Config.ESP.TextColor), math.floor(distance)))
    end
    return table.concat(lines, "\n")
end

local function updateESP(player)
    if player == LocalPlayer then return end
    if not Config.ESP.Enabled then cleanupESPForPlayer(player); return end
    local character = player.Character
    if not character or not character.Parent then cleanupESPForPlayer(player); return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not humanoid or not rootPart then return end
    local distance = (rootPart.Position - Camera.CFrame.Position).Magnitude

    local isWhitelisted = Config.Aim.Whitelist[player.Name] == true
    local useWLColors = isWhitelisted and Config.ESP.WhitelistHighlight.Enabled
    local fillCol, outlineCol
    if useWLColors then
        fillCol = Config.ESP.WhitelistHighlight.FillColor
        outlineCol = Config.ESP.WhitelistHighlight.OutlineColor
    else
        fillCol = Config.ESP.HighlightColor
        outlineCol = Config.ESP.OutlineColor
    end

    if distance < Config.ESP.MaxDistForHighlight then
        if not espHighlights[player] or not espHighlights[player].Parent then
            local h = Instance.new("Highlight")
            h.Name = "VagrantESP_Highlight"; h.Adornee = character
            h.FillColor = fillCol; h.OutlineColor = outlineCol
            h.FillTransparency = 0.5; h.OutlineTransparency = 0
            h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            h.Parent = character; espHighlights[player] = h
        else
            espHighlights[player].Adornee = character
            espHighlights[player].FillColor = fillCol
            espHighlights[player].OutlineColor = outlineCol
        end
    else
        if espHighlights[player] then espHighlights[player]:Destroy(); espHighlights[player] = nil end
    end

    local anyText = Config.ESP.ShowName or Config.ESP.ShowHP or Config.ESP.ShowDistance
    if not anyText then
        if espBillboards[player] then espBillboards[player]:Destroy(); espBillboards[player] = nil end
        return
    end

    if not espBillboards[player] or not espBillboards[player].Parent then
        local b = Instance.new("BillboardGui")
        b.Name = "VagrantESP_Billboard"
        b.Adornee = character:FindFirstChild("Head") or rootPart
        b.Size = UDim2.new(0, 200, 0, 60); b.StudsOffset = Vector3.new(0, 3, 0)
        b.AlwaysOnTop = true; b.MaxDistance = math.huge
        local l = Instance.new("TextLabel")
        l.Name = "ESP_Label"; l.Size = UDim2.new(1, 0, 1, 0)
        l.BackgroundTransparency = 1
        l.TextColor3 = Config.ESP.TextColor
        l.TextStrokeTransparency = 0
        l.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
        l.TextSize = Config.ESP.TextSize
        l.Font = Enum.Font.GothamBold
        l.TextXAlignment = Enum.TextXAlignment.Center
        l.TextYAlignment = Enum.TextYAlignment.Center
        l.RichText = true
        l.Parent = b
        b.Parent = character
        espBillboards[player] = b
    end

    local b = espBillboards[player]
    if b then
        local l = b:FindFirstChild("ESP_Label")
        if l then
            l.TextSize = Config.ESP.TextSize
            l.RichText = true
            l.Text = buildESPLabelText(player, humanoid, distance)
        end

local function getScreenPoint(part)

local function isTargetDirectlyAboveOrBelow()
    if not Config.Aim.VerticalBlockEnabled then return false end
    local myChar = LocalPlayer.Character
    if not myChar then return false end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return false end
    local myPos = myRoot.Position
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer or Config.Aim.Whitelist[player.Name] then continue end
        local char = player.Character
        if not char or not char.Parent then continue end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not hum or not root or hum.Health <= 0 then continue end
        local pos = root.Position
        local dx, dz = pos.X - myPos.X, pos.Z - myPos.Z
        local horizontalDist = math.sqrt(dx * dx + dz * dz)
        local verticalDist = math.abs(pos.Y - myPos.Y)
        if horizontalDist < Config.Aim.VerticalBlockRadius and verticalDist > Config.Aim.VerticalBlockMinHeight then
            return true
        end
    end
    return false
end

local function collectPlayerTargets()
    local list = {}
    local cameraPos = Camera.CFrame.Position
    local cameraLook = Camera.CFrame.LookVector
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer or Config.Aim.Whitelist[player.Name] then continue end
        local character = player.Character
        if not character or not character.Parent then continue end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local bodyPart = getAimPart(character)
        local rootPart = character:FindFirstChild("HumanoidRootPart")
        if not humanoid or not bodyPart or not rootPart then continue end
        if humanoid.Health <= 0 then continue end
        local worldDist = (rootPart.Position - cameraPos).Magnitude
        if worldDist > Config.Aim.MaxDistance then continue end
        local dirUnit = (bodyPart.Position - cameraPos).Unit
        if cameraLook:Dot(dirUnit) < Config.Aim.MinDot then continue end
        table.insert(list, { part = bodyPart, model = character })
    end
    return list
end

local function collectNPCTargets()
    local list = {}
    local cameraPos = Camera.CFrame.Position
    local cameraLook = Camera.CFrame.LookVector
    for _, model in ipairs(Config.Aim.NPCCachedHeads) do
        if model and model.Parent then
            local head = model:FindFirstChild("Head")
            local hum = model:FindFirstChildOfClass("Humanoid")
            local alive = (hum == nil) or (hum.Health > 0)
            if head and head:IsA("BasePart") and alive then
                local worldDist = (head.Position - cameraPos).Magnitude
                if worldDist <= Config.Aim.NPCMaxDistance then
                    local dirUnit = (head.Position - cameraPos).Unit
                    if cameraLook:Dot(dirUnit) >= Config.Aim.MinDot then
                        table.insert(list, { part = head, model = model })
                    end
                end
            end
        end
    end
    return list
end

local function getClosestTargetToCursor()
    local closest, shortDist = nil, Config.Aim.FOVRadius
    local mousePos = UserInputService:GetMouseLocation()

    local candidates = {}
    if Config.Aim.Enabled then
        for _, t in ipairs(collectPlayerTargets()) do table.insert(candidates, t) end
    end
    if Config.Aim.NPCAimEnabled then
        for _, t in ipairs(collectNPCTargets()) do table.insert(candidates, t) end
    end

    for _, t in ipairs(candidates) do
        local screenPos = getScreenPoint(t.part)
        if screenPos then
            local d = (screenPos - mousePos).Magnitude
            if d < shortDist then
                shortDist = d
                closest = t
            end
        end
    end
    return closest
end

    local my = math.clamp(dy * Config.Aim.Smoothness, -Config.Aim.MaxMovePerFrame, Config.Aim.MaxMovePerFrame)
    mousemoverel(mx, my)
end)

local function scanNPCs()
    local found = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and not isModelPlayerCharacter(obj) then

            local hum = obj:FindFirstChildOfClass("Humanoid")
            if head and head:IsA("BasePart") then

                if alive then

                end
            end
        end
    end
    Config.Aim.NPCCachedHeads = found
    return #found
end

local function setForceSpeedEnabled(enabled)
    Config.ForceSpeed.Enabled = enabled
    if enabled then
        setupForceSpeed()
        if not ForceSpeed.Conn then
            ForceSpeed.Conn = RunService.Heartbeat:Connect(function()
                if not Config.ForceSpeed.Enabled then cleanupForceSpeed(); return end
                if not (ForceSpeed.BodyVel and ForceSpeed.BodyVel.Parent and ForceSpeed.Part and ForceSpeed.Part.Parent) then setupForceSpeed() end
                local hum = getHumanoid()
                if not hum or hum.Health <= 0 then
                    if ForceSpeed.BodyVel then ForceSpeed.BodyVel.Velocity = Vector3.zero end
                    return
                end
                local moveDir = hum.MoveDirection
                if moveDir.Magnitude > 0.1 then
                    local dir = Vector3.new(moveDir.X, 0, moveDir.Z)
                    if dir.Magnitude > 0 then
                        local baseSpeed = math.clamp(Config.ForceSpeed.Speed, 16, 30)
                        local speed = baseSpeed
                        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
                            speed = math.min(35, baseSpeed + 5)
                        end
                        ForceSpeed.BodyVel.Velocity = dir.Unit * speed
                    else
                        ForceSpeed.BodyVel.Velocity = Vector3.zero
                    end
                else
                    ForceSpeed.BodyVel.Velocity = Vector3.zero
                end
            end)
        end
    else
        if ForceSpeed.Conn then pcall(function() ForceSpeed.Conn:Disconnect() end); ForceSpeed.Conn = nil end
        cleanupForceSpeed()
    end
end

local function stopSpectate()
    if not Spectate.Enabled then return end
    Spectate.Enabled = false; Spectate.Target = nil
    if Spectate.PrevSubject and Spectate.PrevSubject.Parent then
        Camera.CameraSubject = Spectate.PrevSubject
    else
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then Camera.CameraSubject = hum end
    end
    if Spectate.PrevType then Camera.CameraType = Spectate.PrevType end
end

local function spectateNext(dir)
    local list = getSpectateList()
    if #list == 0 then stopSpectate(); return end
    local idx = 1
    if Spectate.Target then
        for i, p in ipairs(list) do
            if p == Spectate.Target then idx = i + dir; break end
        end
    end
    if idx > #list then idx = 1 end
    if idx < 1 then idx = #list end
    setSpectateTarget(list[idx])
end

            local list = getSpectateList()

                if p == Spectate.Target then table.remove(list, i); break end
            end
            if #list > 0 then setSpectateTarget(list[1]) else stopSpectate() end
        else stopSpectate() end
    else
        Camera.CameraSubject = hum
    end
end)

        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end

function refreshSpectateList()
    for _, child in ipairs(specListFrame:GetChildren()) do
        if not child:IsA("UIListLayout") and not child:IsA("UIPadding") then child:Destroy() end
    end
    local players = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then table.insert(players, p) end
    end
    if #players == 0 then
        local e = Instance.new("TextLabel")
        e.Size = UDim2.new(1, 0, 0, 20); e.BackgroundTransparency = 1
        e.Text = "No players"; e.TextSize = 11
        e.TextColor3 = Color3.fromRGB(170, 170, 170); e.Parent = specListFrame
        addFont(e); return
    end
    for _, p in ipairs(players) do
        local char = p.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local alive = hum and hum.Health > 0
        local isTarget = (Spectate.Target == p)
        local inWl = Config.Aim.Whitelist[p.Name] == true

        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(1, 0, 0, 22); btn.BorderSizePixel = 0
        btn.Text = ""; btn.AutoButtonColor = true; btn.Parent = specListFrame
        btn.BackgroundColor3 = isTarget and Theme.Accent or Theme.Button
        local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, 4); c.Parent = btn

        local nl = Instance.new("TextLabel")
        nl.Size = UDim2.new(1, -110, 1, 0); nl.Position = UDim2.new(0, 8, 0, 0)
        nl.BackgroundTransparency = 1
        nl.Text = (isTarget and "▶ " or "") .. p.Name .. (inWl and "  [WL]" or "")
        nl.TextColor3 = Theme.Text; nl.TextSize = 11
        nl.TextXAlignment = Enum.TextXAlignment.Left; nl.Parent = btn
        addFont(nl)

        local wlBtn = Instance.new("TextButton")
        wlBtn.Size = UDim2.new(0, 32, 0, 18); wlBtn.Position = UDim2.new(1, -100, 0, 2)
        wlBtn.BackgroundColor3 = inWl and Color3.fromRGB(100, 40, 40) or Color3.fromRGB(60, 60, 60)
        wlBtn.BorderSizePixel = 0; wlBtn.Text = inWl and "-WL" or "+WL"
        wlBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        wlBtn.TextSize = 9; wlBtn.Parent = btn
        addFont(wlBtn)
        local wlc2 = Instance.new("UICorner"); wlc2.CornerRadius = UDim.new(0, 4); wlc2.Parent = wlBtn

        wlBtn.MouseButton1Click:Connect(function()
            if Config.Aim.Whitelist[p.Name] then Config.Aim.Whitelist[p.Name] = nil
            else Config.Aim.Whitelist[p.Name] = true end
            refreshSpectateList(); if refreshWhitelistUI then refreshWhitelistUI() end
        end)

        local sl = Instance.new("TextLabel")
        sl.Size = UDim2.new(0, 60, 1, 0); sl.Position = UDim2.new(1, -64, 0, 0)
        sl.BackgroundTransparency = 1
        sl.Text = alive and ("HP " .. math.floor(hum.Health)) or "DEAD"
        sl.TextColor3 = alive and Color3.fromRGB(200, 200, 200) or Color3.fromRGB(200, 100, 100)
        sl.TextSize = 10
        sl.TextXAlignment = Enum.TextXAlignment.Right; sl.Parent = btn
        addFont(sl)

        btn.MouseButton1Click:Connect(function()
            if alive then setSpectateTarget(p); refreshSpectateList() end
        end)
    end
end

refreshWhitelistUI = function()
    for _, child in ipairs(wlListFrame:GetChildren()) do
        if not child:IsA("UIListLayout") and not child:IsA("UIPadding") then child:Destroy() end
    end
    local names = {}
    for n in pairs(Config.Aim.Whitelist) do table.insert(names, n) end
    table.sort(names)
    if #names == 0 then
        local e = Instance.new("TextLabel")
        e.Size = UDim2.new(1, 0, 0, 20); e.BackgroundTransparency = 1
        e.Text = "List is empty"; e.TextSize = 11
        e.TextColor3 = Color3.fromRGB(170, 170, 170); e.Parent = wlListFrame
        addFont(e); return
    end
    for _, name in ipairs(names) do
        local row = Instance.new("Frame")
        row.Size = UDim2.new(1, 0, 0, 20); row.BorderSizePixel = 0; row.Parent = wlListFrame
        addBtn(row)
        local rc = Instance.new("UICorner"); rc.CornerRadius = UDim.new(0, 4); rc.Parent = row

        local inGame = false
        for _, p in ipairs(Players:GetPlayers()) do
            if p.Name == name then inGame = true; break end
        end

        local nl = Instance.new("TextLabel")
        nl.Size = UDim2.new(1, -40, 1, 0); nl.Position = UDim2.new(0, 8, 0, 0)
        nl.BackgroundTransparency = 1
        nl.Text = name .. (inGame and "" or "  (not in game)")
        nl.TextSize = 11
        nl.TextColor3 = inGame and Theme.Text or Color3.fromRGB(170, 170, 170)
        nl.TextXAlignment = Enum.TextXAlignment.Left; nl.Parent = row
        addFont(nl)

        local rb = Instance.new("TextButton")
        rb.Size = UDim2.new(0, 20, 0, 16); rb.Position = UDim2.new(1, -22, 0, 2)
        rb.BackgroundColor3 = Color3.fromRGB(120, 40, 40); rb.BorderSizePixel = 0
        rb.Text = "X"; rb.TextColor3 = Color3.fromRGB(255, 255, 255)
        rb.TextSize = 11; rb.Parent = row
        addFont(rb)
        local rbc = Instance.new("UICorner"); rbc.CornerRadius = UDim.new(0, 4); rbc.Parent = rb

        rb.MouseButton1Click:Connect(function()
            Config.Aim.Whitelist[name] = nil
            refreshWhitelistUI()
            if refreshSpectateList then refreshSpectateList() end
        end)
    end
end

makeTrioBtn(wlBtnRow, "Add All", function()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then Config.Aim.Whitelist[p.Name] = true end
    end
    refreshWhitelistUI(); if refreshSpectateList then refreshSpectateList() end
end)

    if refreshSpectateList then refreshSpectateList() end

    Config.Aim.Whitelist = {}
    refreshWhitelistUI(); if refreshSpectateList then refreshSpectateList() end
end)
