-- ESP
local espHighlights,espBillboards={},{}
local tracerLines={}
local function getOrCreateTracer(p)
    if not tracerLines[p] then
        local l=Drawing.new("Line"); l.Visible=false; l.Color=Config.ESP.TracerColor; l.Thickness=Config.ESP.TracerThickness; l.Transparency=1; l.From=Vector2.new(0,0); l.To=Vector2.new(0,0)
        tracerLines[p]=l
    end
    return tracerLines[p]
end
local function updateTracer(p,char,root,distance)
    local l=getOrCreateTracer(p)
    if not Config.ESP.ShowTracers or distance>=Config.ESP.MaxDistForHighlight then l.Visible=false; return end
    local myChar=LocalPlayer.Character; if not myChar then l.Visible=false; return end
    local myRoot=myChar:FindFirstChild("HumanoidRootPart"); if not myRoot then l.Visible=false; return end
    local target=char:FindFirstChild("Head") or root
    local spF,onF=Camera:WorldToViewportPoint(myRoot.Position); local spT,onT=Camera:WorldToViewportPoint(target.Position)
    if not onF or not onT or spF.Z<=0 or spT.Z<=0 then l.Visible=false; return end
    l.From=Vector2.new(spF.X,spF.Y); l.To=Vector2.new(spT.X,spT.Y); l.Color=Config.ESP.TracerColor; l.Thickness=Config.ESP.TracerThickness; l.Transparency=1; l.Visible=true
end
local function hideTracer(p) local l=tracerLines[p]; if l then l.Visible=false end end
local function destroyTracer(p) local l=tracerLines[p]; if l then pcall(function() l:Remove() end); tracerLines[p]=nil end end
local function cleanupESP(p)
    if espHighlights[p] then espHighlights[p]:Destroy(); espHighlights[p]=nil end
    if espBillboards[p] then espBillboards[p]:Destroy(); espBillboards[p]=nil end
    hideTracer(p)
end
local function colorToRichTag(c) return string.format("rgb(%d,%d,%d)",math.floor(c.R*255+0.5),math.floor(c.G*255+0.5),math.floor(c.B*255+0.5)) end
local function buildESPLabelText(p,hum,dist)
    local lines={}
    local sN=Config.ESP.ShowName or safeModeActive
    local sH=Config.ESP.ShowHP and not safeModeActive
    local sD=Config.ESP.ShowDistance and not safeModeActive
    if sN then table.insert(lines,string.format('<font color="%s">%s</font>',colorToRichTag(Config.ESP.NameColor),p.Name)) end
    if sH then table.insert(lines,string.format('<font color="%s">HP: %d/%d</font>',colorToRichTag(Config.ESP.HPColor),math.floor(hum.Health),math.floor(hum.MaxHealth))) end
    if sD then table.insert(lines,string.format('<font color="%s">Distance: %d studs</font>',colorToRichTag(Config.ESP.DistanceColor),math.floor(dist))) end
    return table.concat(lines,"\n")
end
local function updateESP(p)
    if p==LocalPlayer then return end
    if not Config.ESP.Enabled then cleanupESP(p); return end
    local char=p.Character; if not char or not char.Parent then cleanupESP(p); return end
    local hum=char:FindFirstChildOfClass("Humanoid"); local root=char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then return end
    local dist=(root.Position-Camera.CFrame.Position).Magnitude
    updateTracer(p,char,root,dist)
    if dist<Config.ESP.MaxDistForHighlight then
        if not espHighlights[p] or not espHighlights[p].Parent then
            local h=Instance.new("Highlight"); h.Name="VagrantESP_Highlight"; h.Adornee=char; h.FillColor=Config.ESP.HighlightColor; h.OutlineColor=Config.ESP.OutlineColor; h.FillTransparency=0.5; h.OutlineTransparency=0; h.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop; h.Parent=char
            espHighlights[p]=h
        else
            espHighlights[p].Adornee=char; espHighlights[p].FillColor=Config.ESP.HighlightColor; espHighlights[p].OutlineColor=Config.ESP.OutlineColor
        end
    else
        if espHighlights[p] then espHighlights[p]:Destroy(); espHighlights[p]=nil end
    end
    local anyText=Config.ESP.ShowName or Config.ESP.ShowHP or Config.ESP.ShowDistance or safeModeActive
    if not anyText then
        if espBillboards[p] then espBillboards[p]:Destroy(); espBillboards[p]=nil end
        return
    end
    if not espBillboards[p] or not espBillboards[p].Parent then
        local b=Instance.new("BillboardGui"); b.Name="VagrantESP_Billboard"; b.Adornee=char:FindFirstChild("Head") or root; b.Size=UDim2.new(0,200,0,60); b.StudsOffset=Vector3.new(0,3,0); b.AlwaysOnTop=true; b.MaxDistance=math.huge
        local l=Instance.new("TextLabel",b); l.Name="ESP_Label"; l.Size=UDim2.new(1,0,1,0); l.BackgroundTransparency=1; l.TextColor3=Config.ESP.TextColor; l.TextStrokeTransparency=0; l.TextStrokeColor3=Color3.new(0,0,0); l.TextSize=Config.ESP.TextSize; l.Font=Enum.Font.GothamBold; l.TextXAlignment=Enum.TextXAlignment.Center; l.TextYAlignment=Enum.TextYAlignment.Center; l.RichText=true
        b.Parent=char; espBillboards[p]=b
    end
    local b=espBillboards[p]
    if b then
        local l=b:FindFirstChild("ESP_Label")
        if l then l.TextSize=Config.ESP.TextSize; l.RichText=true; l.Text=buildESPLabelText(p,hum,dist) end
        b.Adornee=char:FindFirstChild("Head") or root
    end
end

local function isTargetDirectlyAboveOrBelow()
    if not Config.Aim.VerticalBlockEnabled then return false end
    local myChar=LocalPlayer.Character; if not myChar then return false end
    local myRoot=myChar:FindFirstChild("HumanoidRootPart"); if not myRoot then return false end
    local myPos=myRoot.Position
    for _,p in ipairs(Players:GetPlayers()) do
        if p==LocalPlayer or Config.Aim.Whitelist[p.Name] then continue end
        local char=p.Character; if not char or not char.Parent then continue end
        local hum=char:FindFirstChildOfClass("Humanoid"); local root=char:FindFirstChild("HumanoidRootPart")
        if not hum or not root or hum.Health<=0 then continue end
        local pos=root.Position; local dx,dz=pos.X-myPos.X,pos.Z-myPos.Z
        local hd=math.sqrt(dx*dx+dz*dz); local vd=math.abs(pos.Y-myPos.Y)
        if hd<Config.Aim.VerticalBlockRadius and vd>Config.Aim.VerticalBlockMinHeight then return true end
    end
    return false
end
local function getClosestPlayerToCursor()
    local closest,shortDist=nil,Config.Aim.FOVRadius
    local mousePos=UserInputService:GetMouseLocation(); local camPos=Camera.CFrame.Position; local camLook=Camera.CFrame.LookVector
    local vp=Camera.ViewportSize; local cx,cy=vp.X/2,vp.Y/2
    local refX,refY
    if UserInputService.MouseBehavior==Enum.MouseBehavior.LockCenter or LocalPlayer.CameraMode==Enum.CameraMode.LockFirstPerson then refX,refY=cx,cy else refX,refY=mousePos.X,mousePos.Y end
    for _,p in ipairs(Players:GetPlayers()) do
        if p==LocalPlayer or Config.Aim.Whitelist[p.Name] then continue end
        local char=p.Character; if not char or not char.Parent then continue end
        local hum=char:FindFirstChildOfClass("Humanoid"); local bp=getAimPart(char); local rp=char:FindFirstChild("HumanoidRootPart")
        if not hum or not bp or not rp then continue end
        if hum.Health<=0 then continue end
        local wd=(rp.Position-camPos).Magnitude; if wd>Config.Aim.MaxDistance then continue end
        local dirU=(bp.Position-camPos).Unit; if camLook:Dot(dirU)<Config.Aim.MinDot then continue end
        local sp=getScreenPoint(bp); if not sp then continue end
        local d=(sp-Vector2.new(refX,refY)).Magnitude
        if d<shortDist then shortDist=d; closest=p end
    end
    return closest
end

-- SAVE/LOAD
local CONFIG_FILE="vagrant_config.txt"
local hasFileAPI=type(writefile)=="function" and type(readfile)=="function" and type(isfile)=="function"
local function serializeValue(v)
    if type(v)=="userdata" and v.R and v.G and v.B then return string.format("Color3.fromRGB(%d,%d,%d)",math.floor(v.R*255+0.5),math.floor(v.G*255+0.5),math.floor(v.B*255+0.5))
    elseif type(v)=="number" then return tostring(v)
    elseif type(v)=="string" then return string.format("%q",v)
    elseif type(v)=="boolean" then return tostring(v)
    elseif type(v)=="table" then
        local parts={"{"}
        for k,val in pairs(v) do
            local key
            if type(k)=="string" and k:match("^[%a_][%w_]*$") then key=k
            elseif type(k)=="number" then key="["..tostring(k).."]"
            else key="["..string.format("%q",tostring(k)).."]" end
            table.insert(parts,key.."="..serializeValue(val))
        end
        table.insert(parts,"}")
        return table.concat(parts,",")
    else return "nil" end
end
local function saveConfig()
    if not hasFileAPI then warn("[Vagrant] writefile/readfile not available."); notify("Vagrant Survival","Save failed - no file API"); return false end
    local data="return "..serializeValue({Config=Config,Theme=Theme})
    local ok=pcall(writefile,CONFIG_FILE,data)
    if ok then print("[Vagrant] Config saved."); notify("Vagrant Survival","Config saved successfully!"); return true
    else warn("[Vagrant] Save failed."); notify("Vagrant Survival","Save failed"); return false end
end
local function deepMerge(target,source)
    for k,v in pairs(source) do
        if type(v)=="table" and type(target[k])=="table" and not (type(v)=="userdata") then deepMerge(target[k],v)
        else target[k]=v end
    end
end
local function loadConfig()
    if not hasFileAPI then warn("[Vagrant] writefile/readfile not available."); notify("Vagrant Survival","Load failed - no file API"); return false end
    local ok1,exists=pcall(isfile,CONFIG_FILE); if not ok1 or not exists then warn("[Vagrant] Config file not found."); notify("Vagrant Survival","Config file not found"); return false end
    local ok2,content=pcall(readfile,CONFIG_FILE); if not ok2 or not content then warn("[Vagrant] Could not read file."); notify("Vagrant Survival","Could not read file"); return false end
    local fn=loadstring or load; local ok3,chunk=pcall(fn,content); if not ok3 or not chunk then warn("[Vagrant] Parse error."); notify("Vagrant Survival","Config parse error"); return false end
    local ok4,data=pcall(chunk); if not ok4 or type(data)~="table" then warn("[Vagrant] Config corrupted."); notify("Vagrant Survival","Config corrupted"); return false end
    if data.Config then deepMerge(Config,data.Config) end
    if data.Theme then deepMerge(Theme,data.Theme) end
    applyTheme(); applyFont(); applyNoShadows(); fovCircle.Radius=Config.Aim.FOVRadius
    if Config.Freecam.Enabled then startFreecam() else stopFreecam() end
    if Config.Cursor.Enabled and Config.Cursor.ImageId~="" then
        applyCursorImage(Config.Cursor.ImageId); if cursorImage then cursorImage.Visible=true end
        if Config.Cursor.HideDefault then pcall(function() UserInputService.MouseIconEnabled=false end) end
    else
        if cursorImage then cursorImage.Visible=false end
        pcall(function() UserInputService.MouseIconEnabled=true end)
    end
    if refreshWhitelistUI then refreshWhitelistUI() end
    if refreshSpectateList then refreshSpectateList() end
    syncAll(); syncFontButtons()
    print("[Vagrant] Config loaded."); notify("Vagrant Survival","Config loaded successfully!")
    return true
end
