local DEFAULT_THEME = {
    Background = Color3.fromRGB(10, 8, 14),
    Panel      = Color3.fromRGB(18, 14, 24),
    Outline    = Color3.fromRGB(80, 50, 120),
    Button     = Color3.fromRGB(30, 24, 40),
    Accent     = Color3.fromRGB(150, 100, 220),
    Text       = Color3.fromRGB(240, 240, 245),
}
local Theme = {}
for k, v in pairs(DEFAULT_THEME) do Theme[k] = v end

local ThemeRefs = { bg = {}, panel = {}, stroke = {}, button = {}, text = {}, accent = {} }
local SyncCallbacks = {}

local function registerSync(fn)
    table.insert(SyncCallbacks, fn)
    fn()
end
local function syncAll()
    for _, fn in ipairs(SyncCallbacks) do fn() end
end

local function addBg(el)     table.insert(ThemeRefs.bg, {el=el, p="BackgroundColor3"});     el.BackgroundColor3 = Theme.Background end
local function addPanel(el)  table.insert(ThemeRefs.panel, {el=el, p="BackgroundColor3"});  el.BackgroundColor3 = Theme.Panel end
local function addStroke(el) table.insert(ThemeRefs.stroke, {el=el, p="Color"});            el.Color = Theme.Outline end
local function addBtn(el)    table.insert(ThemeRefs.button, {el=el, p="BackgroundColor3"}); el.BackgroundColor3 = Theme.Button end
local function addText(el)   table.insert(ThemeRefs.text, {el=el, p="TextColor3"});         el.TextColor3 = Theme.Text end
local function addAccent(el) table.insert(ThemeRefs.accent, {el=el, p="BackgroundColor3"}); el.BackgroundColor3 = Theme.Accent end

local function applyTheme()
    for _, it in ipairs(ThemeRefs.bg) do it.el[it.p] = Theme.Background end
    for _, it in ipairs(ThemeRefs.panel) do it.el[it.p] = Theme.Panel end
    for _, it in ipairs(ThemeRefs.stroke) do it.el[it.p] = Theme.Outline end
    for _, it in ipairs(ThemeRefs.button) do it.el[it.p] = Theme.Button end
    for _, it in ipairs(ThemeRefs.text) do it.el[it.p] = Theme.Text end
    for _, it in ipairs(ThemeRefs.accent) do it.el[it.p] = Theme.Accent end
    syncAll()
end

        cursorImage.Image = ""
        cursorImage.Visible = false
        return
    end
    local cleanId = tostring(id):gsub("%s", "")
    local url
    if cleanId:find("rbxassetid://") then url = cleanId
    elseif cleanId:find("^%d+$") then url = "rbxassetid://" .. cleanId
    else url = cleanId end
    cursorImage.Image = url
    Config.Cursor.ImageId = cleanId
end

local cps_state = {

    onLiveChange = nil,

local function toHex(col)
    return string.format("#%02X%02X%02X",
        math.floor(col.R * 255 + 0.5),
        math.floor(col.G * 255 + 0.5),
        math.floor(col.B * 255 + 0.5))
end

local function fromHex(s)
    s = s:gsub("#", "")
    if #s ~= 6 then return nil end
    local r = tonumber(s:sub(1, 2), 16)
    local g = tonumber(s:sub(3, 4), 16)
    local b = tonumber(s:sub(5, 6), 16)
    if r and g and b then return Color3.fromRGB(r, g, b) end
    return nil
end

local function updateCursorPos()
    local radius = wheelSize / 2
    local angle = cps_state.hue * 2 * math.pi
    local dist = cps_state.sat * radius
    local x = radius + math.cos(angle) * dist
    local y = radius + math.sin(angle) * dist
    wheelCursor.Position = UDim2.new(0, x, 0, y)
    wheelCursor.BackgroundColor3 = cps_state.currentColor
end

local function refreshUI()
    cps_state.currentColor = Color3.fromHSV(cps_state.hue, cps_state.sat, cps_state.val)

    updateCursorPos()

    valueFill.Size = UDim2.new(cps_state.val, 0, 1, 0)
    valueVal.Text = tostring(math.floor(cps_state.val * 100 + 0.5))

    local r, g, b = cps_state.currentColor.R, cps_state.currentColor.G, cps_state.currentColor.B
    rgbSliders["R"].fill.Size = UDim2.new(r, 0, 1, 0)
    rgbSliders["R"].val.Text = tostring(math.floor(r * 255 + 0.5))
    rgbSliders["G"].fill.Size = UDim2.new(g, 0, 1, 0)
    rgbSliders["G"].val.Text = tostring(math.floor(g * 255 + 0.5))
    rgbSliders["B"].fill.Size = UDim2.new(b, 0, 1, 0)
    rgbSliders["B"].val.Text = tostring(math.floor(b * 255 + 0.5))

    cpPreview.BackgroundColor3 = cps_state.currentColor
    local brightness = (r + g + b) / 3
    cpPreviewText.TextColor3 = brightness > 0.5 and Color3.fromRGB(0, 0, 0) or Color3.fromRGB(255, 255, 255)

    local abs = wheelButton.AbsolutePosition
    local size = wheelButton.AbsoluteSize
    local cx = abs.X + size.X / 2
    local cy = abs.Y + size.Y / 2
    local dx = mx - cx
    local dy = my - cy
    local radius = size.X / 2
    local dist = math.sqrt(dx * dx + dy * dy)
    if dist > radius then dist = radius end
    cps_state.sat = dist / radius
    cps_state.hue = (math.atan2(dy, dx) / (2 * math.pi)) % 1
    refreshUI()
end

local espHighlights, espBillboards = {}, {}
local tracerLines = {}

local function getOrCreateTracer(player)
    if not tracerLines[player] then
        local line = Drawing.new("Line")
        line.Visible = false
        line.Color = Config.ESP.TracerColor
        line.Thickness = Config.ESP.TracerThickness
        line.Transparency = 1
        line.From = Vector2.new(0, 0)
        line.To = Vector2.new(0, 0)
        tracerLines[player] = line
    end
    return tracerLines[player]
end

local function updateTracer(player, character, rootPart, distance)
    local line = getOrCreateTracer(player)
    if not Config.ESP.ShowTracers or distance >= Config.ESP.MaxDistForHighlight then
        line.Visible = false; return
    end
    local targetPos = character:FindFirstChild("Head") or rootPart
    local sp, onScreen = Camera:WorldToViewportPoint(targetPos.Position)
    if not onScreen or sp.Z <= 0 then line.Visible = false; return end
    local viewport = Camera.ViewportSize
    line.From = Vector2.new(viewport.X / 2, viewport.Y)
    line.To = Vector2.new(sp.X, sp.Y)
    line.Color = Config.ESP.TracerColor
    line.Thickness = Config.ESP.TracerThickness
    line.Transparency = 1
    line.Visible = true
end

local function hideTracer(player)
    local line = tracerLines[player]
    if line then line.Visible = false end
end

local function destroyTracer(player)
    local line = tracerLines[player]
    if line then pcall(function() line:Remove() end); tracerLines[player] = nil end
end

local function cleanupESPForPlayer(player)
    if espHighlights[player] then espHighlights[player]:Destroy(); espHighlights[player] = nil end
    if espBillboards[player] then espBillboards[player]:Destroy(); espBillboards[player] = nil end
    hideTracer(player)
end

local function colorToRichTag(c)
    return string.format("rgb(%d,%d,%d)",
        math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5))
end

local function buildESPLabelText(player, humanoid, distance)
    local lines = {}
    if Config.ESP.ShowName then
        table.insert(lines, string.format('<font color="%s">%s</font>', colorToRichTag(Config.ESP.NameColor), player.Name))
    end
    if Config.ESP.ShowHP then
        table.insert(lines, string.format('<font color="%s">HP: %d/%d</font>',
            colorToRichTag(Config.ESP.HPColor), math.floor(humanoid.Health), math.floor(humanoid.MaxHealth)))
    end
    if Config.ESP.ShowDistance then
        table.insert(lines, string.format('<font color="%s">Distance: %d studs</font>',
            colorToRichTag(Config.ESP.DistanceColor), math.floor(distance)))
    end
    return table.concat(lines, "\n")
end

local function updateESP(player)
    if player == LocalPlayer then return end
    if not Config.ESP.Enabled then cleanupESPForPlayer(player); return end
    local character = player.Character
    if not character or not character.Parent then cleanupESPForPlayer(player); return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not humanoid or not rootPart then return end
    local distance = (rootPart.Position - Camera.CFrame.Position).Magnitude

    updateTracer(player, character, rootPart, distance)

    if distance < Config.ESP.MaxDistForHighlight then
        if not espHighlights[player] or not espHighlights[player].Parent then
            local h = Instance.new("Highlight")
            h.Name = "VagrantESP_Highlight"; h.Adornee = character
            h.FillColor = Config.ESP.HighlightColor; h.OutlineColor = Config.ESP.OutlineColor
            h.FillTransparency = 0.5; h.OutlineTransparency = 0
            h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            h.Parent = character; espHighlights[player] = h
        else
            espHighlights[player].Adornee = character
            espHighlights[player].FillColor = Config.ESP.HighlightColor
            espHighlights[player].OutlineColor = Config.ESP.OutlineColor
        end
    else
        if espHighlights[player] then espHighlights[player]:Destroy(); espHighlights[player] = nil end
    end

    local anyText = Config.ESP.ShowName or Config.ESP.ShowHP or Config.ESP.ShowDistance
    if not anyText then
        if espBillboards[player] then espBillboards[player]:Destroy(); espBillboards[player] = nil end
        return
    end

    if not espBillboards[player] or not espBillboards[player].Parent then
        local b = Instance.new("BillboardGui")
        b.Name = "VagrantESP_Billboard"
        b.Adornee = character:FindFirstChild("Head") or rootPart
        b.Size = UDim2.new(0, 200, 0, 60); b.StudsOffset = Vector3.new(0, 3, 0)
        b.AlwaysOnTop = true; b.MaxDistance = math.huge

        local l = Instance.new("TextLabel")
        l.Name = "ESP_Label"; l.Size = UDim2.new(1, 0, 1, 0)
        l.BackgroundTransparency = 1
        l.TextColor3 = Config.ESP.TextColor
        l.TextStrokeTransparency = 0
        l.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
        l.TextSize = Config.ESP.TextSize
        l.Font = Enum.Font.GothamBold
        l.TextXAlignment = Enum.TextXAlignment.Center
        l.TextYAlignment = Enum.TextYAlignment.Center
        l.RichText = true
        l.Parent = b
        b.Parent = character
        espBillboards[player] = b
    end

    local b = espBillboards[player]
    if b then
        local l = b:FindFirstChild("ESP_Label")
        if l then
            l.TextSize = Config.ESP.TextSize
            l.RichText = true
            l.Text = buildESPLabelText(player, humanoid, distance)
        end
        b.Adornee = character:FindFirstChild("Head") or rootPart
    end
end

local function setupPlayer(player)
    player.CharacterAdded:Connect(function() task.wait(0.5); updateESP(player) end)
    player.CharacterRemoving:Connect(function() cleanupESPForPlayer(player) end)
    if player.Character then task.wait(0.5); updateESP(player) end
end

    end
end)

local function getAimPart(character)
    if Config.Aim.AimPart == "Head" then
        return character:FindFirstChild("Head")
            or character:FindFirstChild("UpperTorso")
            or character:FindFirstChild("Torso")
            or character:FindFirstChild("HumanoidRootPart")
    else
        return character:FindFirstChild("UpperTorso")
            or character:FindFirstChild("Torso")
            or character:FindFirstChild("LowerTorso")
            or character:FindFirstChild("HumanoidRootPart")
    end
end

local function getScreenPoint(part)
    local sp, on = Camera:WorldToViewportPoint(part.Position)
    if not on or sp.Z <= 0 then return nil end
    return Vector2.new(sp.X, sp.Y)
end

local function isTargetDirectlyAboveOrBelow()
    if not Config.Aim.VerticalBlockEnabled then return false end
    local myChar = LocalPlayer.Character
    if not myChar then return false end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return false end
    local myPos = myRoot.Position
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer or Config.Aim.Whitelist[player.Name] then continue end
        local char = player.Character
        if not char or not char.Parent then continue end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not hum or not root or hum.Health <= 0 then continue end
        local pos = root.Position
        local dx, dz = pos.X - myPos.X, pos.Z - myPos.Z
        local horizontalDist = math.sqrt(dx * dx + dz * dz)
        local verticalDist = math.abs(pos.Y - myPos.Y)
        if horizontalDist < Config.Aim.VerticalBlockRadius and verticalDist > Config.Aim.VerticalBlockMinHeight then
            return true
        end
    end
    return false
end

local function getClosestPlayerToCursor()
    local closest, shortDist = nil, Config.Aim.FOVRadius
    local mousePos = UserInputService:GetMouseLocation()
    local cameraPos = Camera.CFrame.Position
    local cameraLook = Camera.CFrame.LookVector
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer or Config.Aim.Whitelist[player.Name] then continue end
        local character = player.Character
        if not character or not character.Parent then continue end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local bodyPart = getAimPart(character)
        local rootPart = character:FindFirstChild("HumanoidRootPart")
        if not humanoid or not bodyPart or not rootPart then continue end
        if humanoid.Health <= 0 then continue end
        local worldDist = (rootPart.Position - cameraPos).Magnitude
        if worldDist > Config.Aim.MaxDistance then continue end
        local dirUnit = (bodyPart.Position - cameraPos).Unit
        if cameraLook:Dot(dirUnit) < Config.Aim.MinDot then continue end
        local screenPos = getScreenPoint(bodyPart)
        if not screenPos then continue end
        local d = (screenPos - mousePos).Magnitude
        if d < shortDist then shortDist = d; closest = player end
    end
    return closest
end

local hasMouseMoveRel = type(mousemoverel) == "function"

RunService.RenderStepped:Connect(function()
    if not Config.Aim.Enabled or not hasMouseMoveRel then return end
    if isTargetDirectlyAboveOrBelow() then return end
    local target = getClosestPlayerToCursor()
    if not target then return end
    local character = target.Character
    if not character then return end
    local bodyPart = getAimPart(character)
    if not bodyPart then return end
    local screenPos = getScreenPoint(bodyPart)
    if not screenPos then return end
    local mousePos = UserInputService:GetMouseLocation()
    local dx, dy = screenPos.X - mousePos.X, screenPos.Y - mousePos.Y
    if math.abs(dx) < 1 and math.abs(dy) < 1 then return end
    local mx = math.clamp(dx * Config.Aim.Smoothness, -Config.Aim.MaxMovePerFrame, Config.Aim.MaxMovePerFrame)
    local my = math.clamp(dy * Config.Aim.Smoothness, -Config.Aim.MaxMovePerFrame, Config.Aim.MaxMovePerFrame)
    mousemoverel(mx, my)
end)

local Spectate = { Enabled = false, Target = nil, PrevSubject = nil, PrevType = nil }

local function getSpectateList()
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            local char = p.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then table.insert(list, p) end
        end
    end
    return list
end

local function setSpectateTarget(player)
    if not player or player == LocalPlayer then return end
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    if not Spectate.Enabled then
        Spectate.PrevSubject = Camera.CameraSubject
        Spectate.PrevType = Camera.CameraType
    end
    if Freecam and Freecam.Active then
        stopFreecam(); Config.Freecam.Enabled = false; syncAll()
    end
    Spectate.Enabled = true; Spectate.Target = player
    Camera.CameraSubject = hum; Camera.CameraType = Enum.CameraType.Custom
end

local function stopSpectate()
    if not Spectate.Enabled then return end
    Spectate.Enabled = false; Spectate.Target = nil
    if Spectate.PrevSubject and Spectate.PrevSubject.Parent then
        Camera.CameraSubject = Spectate.PrevSubject
    else
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then Camera.CameraSubject = hum end
    end
    if Spectate.PrevType then Camera.CameraType = Spectate.PrevType end
end

local function spectateNext(dir)
    local list = getSpectateList()
    if #list == 0 then stopSpectate(); return end
    local idx = 1
    if Spectate.Target then
        for i, p in ipairs(list) do
            if p == Spectate.Target then idx = i + dir; break end
        end
    end
    if idx > #list then idx = 1 end
    if idx < 1 then idx = #list end
    setSpectateTarget(list[idx])
end

RunService.RenderStepped:Connect(function()
    if not Spectate.Enabled or not Spectate.Target then return end
    local char = Spectate.Target.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not char or not hum or hum.Health <= 0 then
        if Config.Spectate.AutoSwitchOnDeath then
            local list = getSpectateList()
            for i, p in ipairs(list) do
                if p == Spectate.Target then table.remove(list, i); break end
            end
            if #list > 0 then setSpectateTarget(list[1]) else stopSpectate() end
        else stopSpectate() end
    else
        Camera.CameraSubject = hum
    end
end)

local PITCH_LIMIT = math.rad(89)

local Freecam = {
    Active = false, Position = Vector3.new(0, 0, 0), Yaw = 0, Pitch = 0,
    Keys = {}, SavedWalkSpeed = nil, SavedJumpPower = nil,
    PrevCameraType = nil, PrevCameraSubject = nil,
}

local TRACKED_KEYS = {
    [Enum.KeyCode.W] = true, [Enum.KeyCode.A] = true,
    [Enum.KeyCode.S] = true, [Enum.KeyCode.D] = true,
    [Enum.KeyCode.Space] = true, [Enum.KeyCode.LeftControl] = true,
    [Enum.KeyCode.LeftShift] = true,
}

local function freezeLocalCharacter()
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    Freecam.SavedWalkSpeed = hum.WalkSpeed
    Freecam.SavedJumpPower = hum.JumpPower
    hum.WalkSpeed = 0; hum.JumpPower = 0
end

local function unfreezeLocalCharacter()
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    if Freecam.SavedWalkSpeed then hum.WalkSpeed = Freecam.SavedWalkSpeed end
    if Freecam.SavedJumpPower then hum.JumpPower = Freecam.SavedJumpPower end
    Freecam.SavedWalkSpeed = nil; Freecam.SavedJumpPower = nil
end

function startFreecam()
    if Freecam.Active then return end
    if Spectate.Enabled then stopSpectate() end
    Freecam.Active = true
    Freecam.PrevCameraType = Camera.CameraType
    Freecam.PrevCameraSubject = Camera.CameraSubject
    Freecam.Position = Camera.CFrame.Position
    local look = Camera.CFrame.LookVector
    Freecam.Pitch = math.asin(math.clamp(look.Y, -1, 1))
    Freecam.Yaw = math.atan2(-look.X, -look.Z)
    Camera.CameraType = Enum.CameraType.Scriptable
    Camera.CFrame = CFrame.new(Freecam.Position) * CFrame.fromEulerAnglesYXZ(Freecam.Pitch, Freecam.Yaw, 0)
    freezeLocalCharacter()
end

function stopFreecam()
    if not Freecam.Active then return end
    Freecam.Active = false
    unfreezeLocalCharacter()
    UserInputService.MouseBehavior = Enum.MouseBehavior.Default
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        Camera.CameraType = Enum.CameraType.Custom
        Camera.CameraSubject = hum
    elseif Freecam.PrevCameraSubject and Freecam.PrevCameraSubject.Parent then
        Camera.CameraSubject = Freecam.PrevCameraSubject
        Camera.CameraType = Freecam.PrevCameraType or Enum.CameraType.Custom
    end
    for k in pairs(Freecam.Keys) do Freecam.Keys[k] = false end
end

UserInputService.InputBegan:Connect(function(input, gp)
    if TRACKED_KEYS[input.KeyCode] and Freecam.Active then
        Freecam.Keys[input.KeyCode] = true
    end
end)
UserInputService.InputEnded:Connect(function(input)
    if TRACKED_KEYS[input.KeyCode] then Freecam.Keys[input.KeyCode] = false end
end)

RunService.RenderStepped:Connect(function(dt)
    if not Freecam.Active then return end
    local rmbHeld = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
    if rmbHeld then
        if UserInputService.MouseBehavior ~= Enum.MouseBehavior.LockCenter then
            UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
        end
        local delta = UserInputService:GetMouseDelta()
        local sens = Config.Freecam.Sensitivity
        Freecam.Yaw = Freecam.Yaw - delta.X * sens
        Freecam.Pitch = math.clamp(Freecam.Pitch - delta.Y * sens, -PITCH_LIMIT, PITCH_LIMIT)
    else
        if UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter then
            UserInputService.MouseBehavior = Enum.MouseBehavior.Default
        end
    end
    local rotCF = CFrame.fromEulerAnglesYXZ(Freecam.Pitch, Freecam.Yaw, 0)
    local move = Vector3.new()
    if Freecam.Keys[Enum.KeyCode.W] then move = move + rotCF.LookVector end
    if Freecam.Keys[Enum.KeyCode.S] then move = move - rotCF.LookVector end
    if Freecam.Keys[Enum.KeyCode.D] then move = move + rotCF.RightVector end
    if Freecam.Keys[Enum.KeyCode.A] then move = move - rotCF.RightVector end
    if Freecam.Keys[Enum.KeyCode.Space] then move = move + Vector3.new(0, 1, 0) end
    if Freecam.Keys[Enum.KeyCode.LeftControl] then move = move - Vector3.new(0, 1, 0) end
    if move.Magnitude > 0 then
        local speed = 60 * Config.Freecam.Speed
        if Freecam.Keys[Enum.KeyCode.LeftShift] then speed = speed * 3 end
        Freecam.Position = Freecam.Position + move.Unit * speed * dt
    end
    Camera.CFrame = CFrame.new(Freecam.Position) * rotCF
end)

LocalPlayer.CharacterAdded:Connect(function()
    if Freecam.Active then stopFreecam() end
end)

        local v = min + (max - min) * rel

local CONFIG_FILE = "vagrant_config.txt"
local hasFileAPI = type(writefile) == "function" and type(readfile) == "function" and type(isfile) == "function"

local function serializeValue(v)
    if type(v) == "userdata" and v.R and v.G and v.B then
        return string.format("Color3.fromRGB(%d,%d,%d)",
            math.floor(v.R*255+0.5), math.floor(v.G*255+0.5), math.floor(v.B*255+0.5))
    elseif type(v) == "number" then return tostring(v)
    elseif type(v) == "string" then return string.format("%q", v)
    elseif type(v) == "boolean" then return tostring(v)
    elseif type(v) == "table" then
        local parts = {"{"}
        for k, val in pairs(v) do
            local key
            if type(k) == "string" and k:match("^[%a_][%w_]*$") then key = k
            elseif type(k) == "number" then key = "[" .. tostring(k) .. "]"
            else key = "[" .. string.format("%q", tostring(k)) .. "]" end
            table.insert(parts, key .. "=" .. serializeValue(val))
        end
        table.insert(parts, "}")
        return table.concat(parts, ",")
    else return "nil" end
end

local function saveConfig()
    if not hasFileAPI then warn("[Vagrant] writefile/readfile not available."); return false end
    local data = "return " .. serializeValue({ Config = Config, Theme = Theme })
    local ok = pcall(writefile, CONFIG_FILE, data)
    if ok then print("[Vagrant] Config saved.") return true
    else warn("[Vagrant] Save failed."); return false end
end

local function deepMerge(target, source)
    for k, v in pairs(source) do
        if type(v) == "table" and type(target[k]) == "table" and not (type(v) == "userdata") then
            deepMerge(target[k], v)
        else
            target[k] = v
        end
    end
end

local function loadConfig()
    if not hasFileAPI then warn("[Vagrant] writefile/readfile not available."); return false end
    local ok1, exists = pcall(isfile, CONFIG_FILE)
    if not ok1 or not exists then warn("[Vagrant] Config file not found."); return false end
    local ok2, content = pcall(readfile, CONFIG_FILE)
    if not ok2 or not content then warn("[Vagrant] Could not read file."); return false end
    local fn = loadstring or load
    local ok3, chunk = pcall(fn, content)
    if not ok3 or not chunk then warn("[Vagrant] Parse error."); return false end
    local ok4, data = pcall(chunk)
    if not ok4 or type(data) ~= "table" then warn("[Vagrant] Config corrupted."); return false end
    if data.Config then deepMerge(Config, data.Config) end
    if data.Theme then deepMerge(Theme, data.Theme) end

    applyTheme()
    applyFont()
    applyNoShadows()
    fovCircle.Radius = Config.Aim.FOVRadius

    if Config.Freecam.Enabled then startFreecam() else stopFreecam() end

    if Config.Cursor.Enabled and Config.Cursor.ImageId ~= "" then
        cursorImage.Visible = true
        cursorImage.Size = UDim2.new(0, Config.Cursor.Size, 0, Config.Cursor.Size)
        if Config.Cursor.HideDefault then
            pcall(function() UserInputService.MouseIconEnabled = false end)
        end
    else
        cursorImage.Visible = false
        pcall(function() UserInputService.MouseIconEnabled = true end)
    end

    if Config.Cursor.ImageId ~= "" then applyCursorImage(Config.Cursor.ImageId) end
    if refreshWhitelistUI then refreshWhitelistUI() end
    if refreshSpectateList then refreshSpectateList() end

    syncAll()
    syncFontButtons()
    print("[Vagrant] Config loaded.")
    return true
end

function refreshSpectateList()
    for _, child in ipairs(specListFrame:GetChildren()) do
        if not child:IsA("UIListLayout") and not child:IsA("UIPadding") then child:Destroy() end
    end
    local players = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then table.insert(players, p) end
    end
    if #players == 0 then
        local e = Instance.new("TextLabel")
        e.Size = UDim2.new(1, 0, 0, 20); e.BackgroundTransparency = 1
        e.Text = "No players"; e.TextSize = 11
        e.TextColor3 = Color3.fromRGB(150, 150, 150); e.Parent = specListFrame
        addFont(e); return
    end
    for _, p in ipairs(players) do
        local char = p.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local alive = hum and hum.Health > 0
        local isTarget = (Spectate.Target == p)
        local inWl = Config.Aim.Whitelist[p.Name] == true

        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(1, 0, 0, 22); btn.BorderSizePixel = 0
        btn.Text = ""; btn.AutoButtonColor = true; btn.Parent = specListFrame
        btn.BackgroundColor3 = isTarget and Theme.Accent or Theme.Button
        local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, 4); c.Parent = btn

        local nl = Instance.new("TextLabel")
        nl.Size = UDim2.new(1, -110, 1, 0); nl.Position = UDim2.new(0, 8, 0, 0)
        nl.BackgroundTransparency = 1
        nl.Text = (isTarget and "▶ " or "") .. p.Name .. (inWl and "  [WL]" or "")
        nl.TextColor3 = Theme.Text; nl.TextSize = 11
        nl.TextXAlignment = Enum.TextXAlignment.Left; nl.Parent = btn
        addFont(nl)

        local wlBtn = Instance.new("TextButton")
        wlBtn.Size = UDim2.new(0, 32, 0, 18); wlBtn.Position = UDim2.new(1, -100, 0, 2)
        wlBtn.BackgroundColor3 = inWl and Color3.fromRGB(100, 40, 40) or Color3.fromRGB(40, 80, 50)
        wlBtn.BorderSizePixel = 0; wlBtn.Text = inWl and "-WL" or "+WL"
        wlBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        wlBtn.TextSize = 9; wlBtn.Parent = btn
        addFont(wlBtn)
        local wlc2 = Instance.new("UICorner"); wlc2.CornerRadius = UDim.new(0, 4); wlc2.Parent = wlBtn

        wlBtn.MouseButton1Click:Connect(function()
            if Config.Aim.Whitelist[p.Name] then
                Config.Aim.Whitelist[p.Name] = nil
            else
                Config.Aim.Whitelist[p.Name] = true
            end
            refreshSpectateList(); if refreshWhitelistUI then refreshWhitelistUI() end
        end)

        local sl = Instance.new("TextLabel")
        sl.Size = UDim2.new(0, 60, 1, 0); sl.Position = UDim2.new(1, -64, 0, 0)
        sl.BackgroundTransparency = 1
        sl.Text = alive and ("HP " .. math.floor(hum.Health)) or "DEAD"
        sl.TextColor3 = alive and Color3.fromRGB(100, 220, 130) or Color3.fromRGB(200, 100, 100)
        sl.TextSize = 10
        sl.TextXAlignment = Enum.TextXAlignment.Right; sl.Parent = btn
        addFont(sl)

        btn.MouseButton1Click:Connect(function()
            if alive then setSpectateTarget(p); refreshSpectateList() end
        end)
    end
end

refreshSpectateList()
